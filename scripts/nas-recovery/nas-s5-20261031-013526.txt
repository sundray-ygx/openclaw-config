Environment=HOME=/root
Environment=PATH=/usr/local/lib/hermes-agent/venv/bin:/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin

# Synology DSM 用 systemd 219，不给服务注入 INVOCATION_ID，导致 hermes
# 的 supervised-conflict 守卫把自家启动误判为"shell 里重复启动"而死循环。
# 此变量是 hermes 官方出口：声明本进程由外部 supervisor 管理（见 gateway/restart.py）。
Environment=HERMES_GATEWAY_EXTERNAL_SUPERVISOR=1

# Load secrets from Vaultwarden before starting
ExecStartPre=/bin/bash /root/.hermes/scripts/load_secrets.sh

ExecStart=/usr/local/lib/hermes-agent/venv/bin/python -m hermes_cli.main gateway run --replace

Restart=always
RestartSec=5
TimeoutStopSec=30
LimitNOFILE=65536

StandardOutput=journal
StandardError=journal
SyslogIdentifier=hermes-gateway

[Install]
WantedBy=multi-user.target
